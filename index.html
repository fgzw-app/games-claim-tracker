async function registerBackgroundSync() { const reg = await navigator.serviceWorker.ready; if ('periodicSync' in reg) { try { await reg.periodicSync.register('check-games-sync', { minInterval: 6 * 60 * 60 * 1000 // Android checks every 6–12 hours }); console.log('Periodic sync registered successfully!'); } catch (err) { console.log('Periodic sync registration error:', err); } } } // Call after user grants notification permission if (Notification.permission === 'granted') { registerBackgroundSync(); }
